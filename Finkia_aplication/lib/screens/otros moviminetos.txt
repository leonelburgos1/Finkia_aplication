class CustomBottomNavBar extends StatefulWidget {
  const CustomBottomNavBar({super.key});

  @override
  State<CustomBottomNavBar> createState() => _CustomBottomNavBarState();
}

class _CustomBottomNavBarState extends State<CustomBottomNavBar> {
  int selectedIndex = 0;
  final int itemCount = 4; // <-- cambia si tienes más/menos íconos

  @override
  Widget build(BuildContext context) {
    return SizedBox(
      height: 90,
      child: LayoutBuilder(builder: (context, constraints) {
        final totalWidth = constraints.maxWidth;
        final itemWidth = totalWidth / itemCount;

        // notchWidth proporcional al itemWidth (no fijo)
        final notchWidth = min(itemWidth * 0.8, 150.0);
        // Posición izquierda objetivo para la notch (basado en layout real)
        final targetLeft = (selectedIndex * itemWidth) + (itemWidth / 2) - (notchWidth / 2);

        return Stack(
          clipBehavior: Clip.none,
          children: [
            // Fondo del bottom bar (simple rectángulo blanco con sombra)
            Positioned(
              left: 0,
              right: 0,
              bottom: 0,
              child: Container(
                height: 70,
                decoration: BoxDecoration(
                  color: Colors.white,
                  boxShadow: [
                    BoxShadow(color: Colors.black.withOpacity(0.12), blurRadius: 6, offset: const Offset(0, -2)),
                  ],
                ),
              ),
            ),

            // Notch (pieza independiente) animada horizontalmente con AnimatedPositioned
            AnimatedPositioned(
              duration: const Duration(milliseconds: 350),
              curve: Curves.easeOutCubic,
              left: targetLeft.clamp(8.0, totalWidth - notchWidth - 8.0),
              bottom: 30, // sobresale por encima del fondo (ajusta para "encajar")
              child: SizedBox(
                width: notchWidth,
                height: 70,
                child: CustomPaint(
                  painter: _NotchPainter(),
                ),
              ),
            ),

            // Botón circular (ícono flotante) ubicado centrado sobre la notch
            AnimatedPositioned(
              duration: const Duration(milliseconds: 350),
              curve: Curves.easeOutBack,
              left: (targetLeft + notchWidth / 2 - 28).clamp(0.0, totalWidth - 56.0),
              bottom: 24, // la altura sobre la barra; ajusta para que "encaje" en el hueco
              child: GestureDetector(
                onTap: () {
                  // acción central (puedes cambiar)
                },
                child: Container(
                  width: 56,
                  height: 56,
                  decoration: BoxDecoration(
                    color: const Color.fromARGB(255, 255, 197, 15),
                    shape: BoxShape.circle,
                    boxShadow: [BoxShadow(color: Colors.black.withOpacity(0.15), blurRadius: 6, offset: const Offset(0, 3))],
                  ),
                  child: const Icon(Icons.merge_sharp, color: Colors.black),
                ),
              ),
            ),

            // Íconos (fila) — usamos Row y GestureDetector; los íconos "flotan" con transform cuando se seleccionan
            Positioned.fill(
              child: Align(
                alignment: Alignment.bottomCenter,
                child: SizedBox(
                  height: 70,
                  child: Row(
                    children: List.generate(itemCount, (index) {
                      // divide el ancho equitativamente, centra cada ícono en su slot
                      return SizedBox(
                        width: itemWidth,
                        child: Center(child: _buildNavIconByIndex(index)),
                      );
                    }),
                  ),
                ),
              ),
            ),
          ],
        );
      }),
    );
  }

  Widget _buildNavIconByIndex(int index) {
    final isSelected = index == selectedIndex;
    IconData icon;
    switch (index) {
      case 0:
        icon = Icons.home;
        break;
      case 1:
        icon = Icons.restaurant_menu;
        break;
      case 2:
        icon = Icons.bookmark;
        break;
      default:
        icon = Icons.notifications;
    }

    return GestureDetector(
      behavior: HitTestBehavior.translucent,
      onTap: () => setState(() => selectedIndex = index),
      child: AnimatedContainer(
        duration: const Duration(milliseconds: 300),
        transform: Matrix4.translationValues(0, isSelected ? -18 : 0, 0), // "flota" sin usar margin negativo
        curve: Curves.easeOut,
        child: Container(
          padding: const EdgeInsets.all(6),
          decoration: isSelected
              ? BoxDecoration(
                  color: Colors.white,
                  shape: BoxShape.circle,
                  boxShadow: [BoxShadow(color: Colors.black.withOpacity(0.12), blurRadius: 6, offset: const Offset(0, 3))],
                )
              : null,
          child: Icon(
            icon,
            size: isSelected ? 34 : 28,
            color: isSelected ? const Color(0xFFFFC50F) : Colors.black87,
          ),
        ),
      ),
    );
  }
}

/// NotchPainter: dibuja una muesca / hueco suave en la parte superior de su rectángulo
class _NotchPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()..color = Colors.white;

    final double w = size.width;
    final double h = size.height;

    // Dibujamos un notch suave usando curvas; ancho y profundidad relativos al tamaño del widget
    final Path path = Path();
    path.moveTo(0, h * 0.5);
    path.quadraticBezierTo(w * 0.12, h * 0.1, w * 0.25, h * 0.06);
    path.quadraticBezierTo(w * 0.45, 0, w * 0.5, 0);
    path.quadraticBezierTo(w * 0.55, 0, w * 0.75, h * 0.06);
    path.quadraticBezierTo(w * 0.88, h * 0.1, w, h * 0.5);
    path.lineTo(w, h);
    path.lineTo(0, h);
    path.close();

    canvas.drawShadow(path, Colors.black.withOpacity(0.12), 6, true);
    canvas.drawPath(path, paint);
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

// ------------------ CUSTOM BOTTOM NAV BAR ------------------
class CustomBottomNavBar extends StatefulWidget {
  const CustomBottomNavBar({super.key});

  @override
  State<CustomBottomNavBar> createState() => _CustomBottomNavBarState();
}

class _CustomBottomNavBarState extends State<CustomBottomNavBar> {
  int selectedIndex = 0; // 🔹 Guarda cuál icono está activo

  @override
  Widget build(BuildContext context) {
    final Size size = MediaQuery.of(context).size;

    return SizedBox(
      width: size.width,
      height: 80,
      child: Stack(
        children: [
          CustomPaint(
            size: Size(size.width, 80),
            painter: BNBCustomPainter(),
          ),
          Center(
            heightFactor: 0.6,
            child: FloatingActionButton(
              onPressed: () {},
              backgroundColor: const Color.fromARGB(255, 255, 197, 15),
              child: const Icon(Icons.merge_sharp, color: Colors.black),
              elevation: 0.1,
            ),
          ),
          SizedBox(
            width: size.width,
            height: 80,
            child: Row(
              mainAxisAlignment: MainAxisAlignment.spaceEvenly,
              children: [
                _buildNavIcon(Icons.home, 0),
                _buildNavIcon(Icons.restaurant_menu, 1),
                SizedBox(width: size.width * .20),
                _buildNavIcon(Icons.bookmark, 2),
                _buildNavIcon(Icons.notifications, 3),
              ],
            ),
          ),
        ],
      ),
    );
  }

  // 🔹 Método que crea cada ícono
  Widget _buildNavIcon(IconData icon, int index) {
    final bool isSelected = selectedIndex == index;
    return IconButton(
      onPressed: () {
        setState(() => selectedIndex = index);
      },
      icon: Icon(
        icon,
        color: isSelected ? Colors.green : Colors.black, // Cambia color
        size: isSelected ? 32 : 28, // Pequeño efecto visual opcional
      ),
    );
  }
}
